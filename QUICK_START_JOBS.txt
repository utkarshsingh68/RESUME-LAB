💻 PYTHON CODE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

from core.job_fetcher import JobFetcher

fetcher = JobFetcher()

# Fetch LinkedIn jobs
jobs = fetcher.fetch_jsearch_jobs(
    query="python developer",
    location="New York",
    remote_jobs_only=False,
    limit=50
)

# Save to database
fetcher.save_jobs_to_file(jobs, append=True)

print(f"Fetched {len(jobs)} jobs!")
for job in jobs[:5]:
    print(f"  • {job['title']} at {job['company']}")

Error: "RAPIDAPI_KEY not configured"
  → Add RAPIDAPI_KEY to your .env file
  → Restart the backend server

Error: "No jobs fetched"
  → Verify API key is valid
  → Check you haven't exceeded free tier (500/month)
  → Try a different query
  → Check RapidAPI dashboard: https://rapidapi.com/developer/billing

Jobs not showing specific source:
  → Use source=linkedin, source=indeed, or source=glassdoor
  → source=jsearch aggregates all sources
